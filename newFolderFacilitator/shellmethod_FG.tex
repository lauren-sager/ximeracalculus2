\textbf{Student Learning Outcomes}\\
Students should be able to:
\begin{itemize}
    \item Set up integrals to compute volumes of solids of revolution using the shell method.
    \item Compare and contrast the shell method and the disk/washer method.
\end{itemize}
\textbf{Prerequisites:} Students should know the formulas for the shell and disk/washer methods, and be able to visualize solids of revolution to set up the integrals. 
\textbf{Purpose: }To explore the volumes of revolution. This activity focuses on building intuition and confidence with the volume of revolution construction discussed in lecture. You will work with your groups to find volumes of solids of rotation using cylindrical shells.\\

\begin{exercise}
    Find the volume of the solid generated by $g(x)=\frac{x}{\sqrt{1+x^3}}$ from $[1,4]$ rotated about the $y$-axis.
    \textcolor{blue}{{Shell method is simplest in this situation, as the integral will be in terms of $x$. \\
$V=\displaystyle \int_1^4 2\pi x\frac{x}{\sqrt{1+x^3}}dx=2\pi\int_1^4 \frac{x^2}{\sqrt{x^3+1}}dx$\\
$u=1+x^3$. $\frac{du}{dx}=3x^2$, and $u(1)=2$, $u(4)=65$\\
$2\pi \displaystyle \int_2^{65}\frac{1}{3}u^{-1/2}du=\frac{4\pi}{3}u^{1/2}\bigg|_2^{65}=\frac{4\pi}{3}\sqrt{65}-\frac{4\pi}{3}\sqrt{2}$}}
\end{exercise}

\begin{exercise}
    Find the volume of the solid generated by rotating the area enclosed by $f(x)=6-x$, $g(x)=3x-2$ and $x=0$ rotated about the $y$-axis.  
    \begin{enumerate}
        \item Calculate the area using the shell method.\\
        \textcolor{blue}{$V=\displaystyle \int_{\frac23}^2 2\pi x (3x-2)dx+\int_2^62\pi x(6-x)dx$\\
$\displaystyle 2\pi \int_{\frac23}^23x^2-2xdx+2\pi \int_2^6 6x-x^2dx=2\pi(x^3-x^2)\bigg|_{\frac23}^2+2\pi(3x^2-\frac{x^3}{3})\bigg|_2^6$\\
$=2\pi(2^3-2^2-(\frac23)^3+(\frac23)^2)+2\pi(3(6)^2-\frac{6^3}{3}-3(2)^2+\frac{2^3}{3})$\\}
        \item Calculate the area using the disk method.\\
        \textcolor{blue}{$f(x)=6-x \rightarrow x=6-y$ and $g(x)=3x-2 \rightarrow \frac{y}{3}+\frac23=x$\\
$V=\displaystyle \int_0^4 \pi(6-y)^2-\pi(\frac{y}{3}+\frac23)^2dy=\pi \int_0^4 36-12y+y^2-\frac{y^2}{9}-\frac{4}{3}y -\frac{4}{9}dy$\\
$\displaystyle=\int_0^4 \frac{8}{9}y^2-\frac{40}{3}y-\frac{320}{9}dy=\frac{8}{27}y^3-\frac{20}{3}y^2-\frac{320}{9}y\bigg|_0^4$\\}
        \item Reflect on the two methods. Which method is easier for this problem? Why?\\
        \textcolor{orange}{For this particular example, this is a somewhat subjective question. It is helpful to remind students that for some problems (for example, the next exercise), one method or the other may be needed.}
    \end{enumerate}
\end{exercise}

\begin{exercise}
    Find the volume of the solid generated by rotating the area under the function $f(x)=4x^{-1}$ on $[1,4]$ around the $y$-axis.  (Hint: you may need multiple integrals.) \\
    \textcolor{blue}{$y=\frac4x$ or $x=\frac4y$\\
$V=\displaystyle\int_1^4\pi(4y^{-1})^2-\pi(1)^2dy+\int_0^1\pi(4)^2-\pi(1)^2dy$\\
$=\displaystyle\pi \int_1^4 16y^{-2}-1dy+\int_0^1 15\pi=\pi(-16y^{-1}-y)\bigg|_1^4+15\pi x\bigg|_0^1=\pi(-4-4)-\pi(-16-1)+15\pi=-8\pi+17\pi+15\pi=24\pi$\\
or
$V=\displaystyle \int_1^42\pi x\frac{4}{x}dx=\int_1^4 8\pi dx=8\pi x\bigg|_1^4=8\pi(4-1)=24\pi$}
\end{exercise}
